// ── State ──────────────────────────────────────────────

let state = {

  friends: ['Alice', 'Bob', 'Charlie'],

  expenses: [

    {id:1,desc:'Dinner',amount:60,paidBy:'You',cat:'🍽️',members:['You','Alice','Bob'],splitType:'equal',date:'2026-05-25',notes:''},

    {id:2,desc:'Uber home',amount:18,paidBy:'Alice',cat:'🚗',members:['You','Alice'],splitType:'equal',date:'2026-05-24',notes:''},

    {id:3,desc:'Groceries',amount:45,paidBy:'Bob',cat:'🛒',members:['You','Alice','Bob','Charlie'],splitType:'equal',date:'2026-05-23',notes:'Split evenly'},

    {id:4,desc:'Cinema tickets',amount:32,paidBy:'You',cat:'🎉',members:['You','Charlie'],splitType:'equal',date:'2026-05-22',notes:''},

  ],

  currency: '£',

  theme: 'light',

  nextId: 5

};

let splitType = 'equal';

let selectedMembers = [];


// ── Persistence ────────────────────────────────────────

function save() { localStorage.setItem('splitofriend', JSON.stringify(state)); }

function load() {

  const d = localStorage.getItem('splitofriend');

  if (d) state = JSON.parse(d);

  document.documentElement.dataset.theme = state.theme;

  document.getElementById('currency-sel').value = state.currency;

  updateThemeIcon();

}


// ── Helpers ────────────────────────────────────────────

const colors = [

  ['#E1F5EE','#0F6E56'],['#E6F1FB','#185FA5'],['#FAEEDA','#854F0B'],

  ['#FBEAF0','#993556'],['#EEEDFE','#534AB7'],['#EAF3DE','#3B6D11']

];

function avatarStyle(name) {

  const i = (name.charCodeAt(0) + (name.charCodeAt(1)||0)) % colors.length;

  return `background:${colors[i][0]};color:${colors[i][1]}`;

}

function initials(name) { return name.split(' ').map(w=>w[0]).join('').toUpperCase().slice(0,2); }

function fmt(n) { return state.currency + Math.abs(n).toFixed(2); }

function allMembers() { return ['You', ...state.friends]; }

function today() { return new Date().toISOString().slice(0,10); }

function fmtDate(d) {

  if (!d) return '';

  const t = today(), y = new Date(); y.setDate(y.getDate()-1);

  const yy = y.toISOString().slice(0,10);

  if (d === t) return 'Today';

  if (d === yy) return 'Yesterday';

  return new Date(d).toLocaleDateString('en-GB',{day:'numeric',month:'short'});

}


function computeBalances() {

  const bal = {};

  allMembers().forEach(m => bal[m] = 0);

  state.expenses.forEach(e => {

    if (!e.members || !e.members.length) return;

    const share = e.amount / e.members.length;

    e.members.forEach(m => { if (m !== e.paidBy) bal[m] = (bal[m]||0) - share; });

    const others = e.members.filter(m => m !== e.paidBy);

    bal[e.paidBy] = (bal[e.paidBy]||0) + others.length * share;

  });

  return bal;

}


function toast(msg) {

  const t = document.getElementById('toast');

  t.textContent = msg;

  t.classList.add('show');

  setTimeout(() => t.classList.remove('show'), 2000);

}


// ── Theme & Currency ───────────────────────────────────

function toggleTheme() {

  state.theme = state.theme === 'light' ? 'dark' : 'light';

  document.documentElement.dataset.theme = state.theme;

  updateThemeIcon();

  save();

}

function updateThemeIcon() {

  document.getElementById('theme-btn').innerHTML = state.theme === 'dark'

    ? '<i class="ti ti-sun"></i>' : '<i class="ti ti-moon"></i>';

}

function setCurrency(c) { state.currency = c; save(); renderAll(); }


// ── Tabs ───────────────────────────────────────────────

function switchTab(t) {

  ['dashboard','add','friends','settle','history'].forEach(id => {

    document.getElementById('tab-'+id).classList.toggle('active', id===t);

    document.getElementById('t-'+id).classList.toggle('active', id===t);

  });

  if(t==='dashboard') renderDashboard();

  if(t==='add') renderAddForm();

  if(t==='friends') renderFriends();

  if(t==='settle') renderSettle();

  if(t==='history') renderHistory();

}

function renderAll() {

  renderDashboard(); renderFriends(); renderSettle(); renderHistory();

}


// ── Dashboard ──────────────────────────────────────────

function renderDashboard() {

  const bal = computeBalances();

  const youOwe = Math.max(0, -(bal['You']||0));

  const youOwed = Math.max(0, bal['You']||0);

  const total = state.expenses.reduce((s,e)=>s+e.amount,0);

  document.getElementById('s-total').textContent = fmt(total);

  document.getElementById('s-owed').textContent = fmt(youOwed);

  document.getElementById('s-owe').textContent = fmt(youOwe);


  // Balances

  const bl = document.getElementById('d-balances');

  if (!state.friends.length) { bl.innerHTML = '<div class="empty"><i class="ti ti-users"></i>Add friends to see balances</div>'; }

  else {

    bl.innerHTML = state.friends.map(f => {

      const b = bal[f] || 0;

      const label = b > 0 ? 'owes you' : b < 0 ? 'you owe' : 'settled up';

      const cls = b > 0 ? 'green-text' : b < 0 ? 'red-text' : '';

      const sign = b > 0 ? '+' : b < 0 ? '-' : '';

      return `<div class="list-item">

        <div class="avatar" style="${avatarStyle(f)}">${initials(f)}</div>

        <div class="list-info"><div class="list-name">${f}</div><div class="list-sub">${label}</div></div>

        <div class="amount-right"><div class="amount-big ${cls}">${sign}${fmt(b)}</div></div>

      </div>`;

    }).join('');

  }


  // Expenses

  const el = document.getElementById('d-expenses');

  if (!state.expenses.length) { el.innerHTML = '<div class="empty"><i class="ti ti-receipt"></i>No expenses yet</div>'; return; }

  el.innerHTML = state.expenses.slice().reverse().slice(0,4).map(e => expenseRow(e)).join('');

}


function expenseRow(e) {

  const share = (e.amount / (e.members||[e.paidBy]).length).toFixed(2);

  return `<div class="list-item" onclick="openDetail(${e.id})" style="cursor:pointer">

    <div class="cat-icon">${e.cat}</div>

    <div class="list-info">

      <div class="list-name">${e.desc}</div>

function toggleMember(el, name) {

  el.classList.toggle('selected');

  if (el.classList.contains('selected')) selectedMembers.push(name);

  else selectedMembers = selectedMembers.filter(m => m !== name);

}


function setSplit(type) {

  splitType = type;

  document.getElementById('so-equal').classList.toggle('selected', type==='equal');

  document.getElementById('so-payer').classList.toggle('selected', type==='payer');

}


function addExpense() {

  const desc = document.getElementById('e-desc').value.trim();

  const amount = parseFloat(document.getElementById('e-amount').value);

  const cat = document.getElementById('e-cat').value;

  const paidBy = document.getElementById('e-paidby').value;

  const date = document.getElementById('e-date').value || today();

  const notes = document.getElementById('e-notes').value.trim();

  if (!desc) { toast('⚠️ Please enter a description'); return; }

  if (!amount || amount <= 0) { toast('⚠️ Please enter a valid amount'); return; }

  if (!selectedMembers.length) { toast('⚠️ Select at least one person'); return; }

  state.expenses.push({ id: state.nextId++, desc, amount, paidBy, cat, members: selectedMembers.slice(), splitType, date, notes });

  save();

  document.getElementById('e-desc').value = '';

  document.getElementById('e-amount').value = '';

  document.getElementById('e-notes').value = '';

  toast('✅ Expense added!');

  switchTab('dashboard');

}


// ── Friends ────────────────────────────────────────────

function addFriend() {

  const inp = document.getElementById('f-name');

  const name = inp.value.trim();

  if (!name) return;

  if (state.friends.includes(name) || name === 'You') { toast('⚠️ Already in your group'); return; }

  state.friends.push(name);

  inp.value = '';

  save();

  renderFriends();

  toast('👋 ' + name + ' added!');

}


function removeFriend(name) {

  if (!confirm('Remove ' + name + '? Their expenses will remain.')) return;

  state.friends = state.friends.filter(f => f !== name);

  save();

  renderFriends();

  renderDashboard();

}


function renderFriends() {

  const fl = document.getElementById('friends-list');

  if (!state.friends.length) { fl.innerHTML = '<div class="empty"><i class="ti ti-user-plus"></i>No friends yet. Add one above!</div>'; return; }

  const bal = computeBalances();

  fl.innerHTML = ['You', ...state.friends].map(f => {

    const b = bal[f] || 0;

    const cls = b > 0 ? 'green' : b < 0 ? 'red' : '';

    const label = b > 0 ? 'gets back ' + fmt(b) : b < 0 ? 'owes ' + fmt(b) : 'settled';

    const del = f !== 'You' ? `<button class="btn sm danger" onclick="removeFriend('${f}')"><i class="ti ti-trash"></i></button>` : '';

    return `<div class="list-item">

      <div class="avatar" style="${avatarStyle(f)}">${initials(f)}</div>

      <div class="list-info"><div class="list-name">${f}${f==='You'?' (you)':''}</div><div class="list-sub">${label}</div></div>

      ${del}

    </div>`;

  }).join('');

}


// ── Settle ─────────────────────────────────────────────

function renderSettle() {

  const bal = computeBalances();

  const owers = Object.entries(bal).filter(([,v])=>v<0).map(([k,v])=>({name:k,amount:-v}));

  const oweds = Object.entries(bal).filter(([,v])=>v>0).map(([k,v])=>({name:k,amount:v}));

  const sl = document.getElementById('settle-list');

  if (!owers.length) { sl.innerHTML = '<div class="empty"><i class="ti ti-circle-check"></i>Everyone is settled up! 🎉</div>'; }

  else {

    const settlements = [];

    const o1 = owers.map(x=>({...x})), o2 = oweds.map(x=>({...x}));

    while (o1.length && o2.length) {

      const a = o1[0], b = o2[0];

      const amt = Math.min(a.amount, b.amount);

      settlements.push({ from: a.name, to: b.name, amount: amt });

      a.amount -= amt; b.amount -= amt;

      if (a.amount < 0.01) o1.shift();

      if (b.amount < 0.01) o2.shift();

    }

    sl.innerHTML = settlements.map(s => `<div class="settle-row">

      <div class="avatar sm" style="${avatarStyle(s.from)}">${initials(s.from)}</div>

      <div style="flex:1;font-size:13px"><b>${s.from}</b> pays <b>${s.to}</b></div>

      <div class="settle-amt">${fmt(s.amount)}</div>

      <div class="avatar sm" style="${avatarStyle(s.to)}">${initials(s.to)}</div>

    </div>`).join('');

  }


  const ps = document.getElementById('person-summary');

  ps.innerHTML = allMembers().map(m => {

    const paid = state.expenses.filter(e=>e.paidBy===m).reduce((s,e)=>s+e.amount,0);

    const share = state.expenses.filter(e=>(e.members||[]).includes(m)).reduce((s,e)=>s+e.amount/(e.members||[m]).length,0);

    const net = bal[m] || 0;

    const cls = net >= 0 ? 'green-text' : 'red-text';

    const sign = net >= 0 ? '+' : '-';

    return `<div class="list-item">

      <div class="avatar" style="${avatarStyle(m)}">${initials(m)}</div>

      <div class="list-info">

        <div class="list-name">${m}</div>

        <div class="list-sub">Paid ${fmt(paid)} · Share ${fmt(share)}</div>

      </div>

      <div class="amount-right"><div class="amount-big ${cls}">${sign}${fmt(net)}</div></div>

    </div>`;

  }).join('');

}


// ── History ────────────────────────────────────────────

function renderHistory() {

  const hl = document.getElementById('history-list');

  if (!state.expenses.length) { hl.innerHTML = '<div class="empty"><i class="ti ti-history"></i>No expenses yet</div>'; return; }

  const sorted = state.expenses.slice().sort((a,b)=>(b.date||'').localeCompare(a.date||''));

  let lastDate = '', html = '';

  sorted.forEach(e => {

    if (e.date !== lastDate) {

      html += `<div class="section-heading">${fmtDate(e.date)}</div>`;

      lastDate = e.date;

    }

    html += expenseRow(e);

  });

  hl.innerHTML = html;

}

}
